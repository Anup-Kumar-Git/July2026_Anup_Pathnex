	🔹 Ansible — Install Packages Using Variables

	---
	  - name: Install package for Pathnex using variables
	    hosts: all
	    become:yes

	  vars:
	      pathnex_package:
		  - tree
		   - unzip
		   - vim


	  tasks:
	 	- name: Install required packages
		  yum:
			name: "{{ pathnex_package}}"
			state: present

	🔹 Kubernetes — Add Liveness & Readiness Probes


	apiVersion : v1
	kind: pod
	metadata:
		name: pathnex-probe-pod
	spec:
		containers:
		   -name:web
		    image: nginx
		     livenessProbe:
			   httpGet:
				path: /
				port: 80
			   initialDelaySecond: 5
			readinessProbe:
				httpGet:
					path: /
					port:80
				initialDelaySecond:3

# Conditional Stages------------------

	🔹 Jenkins Pipeline — Conditional Execution
		learn how to **run stages based on branch or parameter conditions**.

	pipeline{
		agent any 
		parameters {
			string(name: 'ENVIRONMENT', defaultValue: 'dev', description: 'Deployment Environment')
		}
		environnment {
			Insitute_Name ="Pathnex"
		}
		
		stages {
			stage('checkout') {
				steps{
					 git branch: 'main', url: 'https://github.com/Pathnex/sample-java-app.git'
				}
			}
			stage('build') {
				when {
					branch "main"
				}
				steps {
					sh 'mvn clean compile'		
				}
			}		
			stage('Deploy') {
				when {
					expression {params.ENIVORMENT=='dev'}
				}
				steps {
					sh 'echo "Deploying to $Enivorment by Insitute_Name"'
				}
			}
		}
	}




	🔹 GitLab CI — Conditional Jobs

		learn how to **run jobs based on variables or branches**.

	stages:
		- build
		- deploy

	build:
		stage:build
		image:mavan:3.8.1-jdk-17
		script:
			- git clone  https://github.com/Pathnex/sample-java-app.git
			- cd sample-java-app
			- mvn clean compile
		only:
			- main

	deploy:
		stage: deploy
		image:alpine:latest
		script:
			- echo "Deploy to $CI_Enviroment_name by Pathnex"
		only:
			variables:
				- $CI_Enivorment_Name=="dev"
