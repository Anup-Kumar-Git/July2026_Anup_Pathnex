# Parallel Builds-------------
	🔹 Jenkins Pipeline — Parallel Build & Test
		how to **run build and test in parallel**.

	pipeline{
	   agent any{
		stages{
			stage('Parallel Task') {
			   paralell {
				stage('Build') {
					steps {
					   sh 'mvn clean compile'
					}
				}
				stage('Test') {
					steps {
						sh 'mvn test'
					}
				}
			}
		}
	}				
	}	
     }
 

	🔹 GitLab CI — Parallel Jobs
		how to **run parallel jobs in GitLab CI**.

	stages:
	   - build-test
 
	maven-build:
		stage: build-test
 		image: mven:3.8.1-jdk-17
		script:
		    - git clone https://github.com/Pathnex/sample-java-app.git
		    - cd sample-java-app
		    - mvn clean compile

	mavan-test:
		stage: build-test
